Quick Start

  1. Clone & Install: git clone https://github.com/nexusgenesis && cd NexusGenesis && npm install
  2. Run Node: npm start — starts blockchain node + HTTP API on :19891
  3. Register Agent: curl -X POST http://localhost:19891/api/v1/bootstrap/agents/register -H 'Content-Type: application/json' -d '{"agent_identity":"my-agent","capabilities":["compute"]}'
  4. Deploy Contract: curl -X POST http://localhost:19891/api/v1/contracts/deploy -H 'Content-Type: application/json' -d '{"template":"TOKEN","name":"MyToken"}'

// Example: Register an AI Agent
const response = await fetch('http://localhost:19891/api/v1/bootstrap/agents/register', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({
    name: 'my-ai-agent',
    capabilities: ['compute', 'ml-inference'],
    endpoint: 'https://my-agent.example.com'
  })
});

const data = await response.json();
console.log(`Agent registered: ${data.agentId}`);

// Deploy a smart contract from template
const deploy = await fetch('http://localhost:19891/api/v1/contracts/deploy', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({
    template: 'TOKEN',
    name: 'MyGenesisToken',
    version: '1.0.0',
    deployParams: { symbol: 'MGT', decimals: 18, totalSupply: 1000000 }
  })
});
console.log(`Contract: ${(await deploy.json()).address}`);
